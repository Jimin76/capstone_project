let
    // ✅ 외부 쿼리 안전하게 캐싱
    PBI_Fix = PBI,
    WD_Fix = WD_PBI,

    // Load source data
    Source = Excel.Workbook(File.Contents("C:\\Users\\ADMIN\\Downloads\\Reporte Febrero 2025-Rawdata.xlsx"), null, true),
    RawData = Source{[Item="Data",Kind="Sheet"]}[Data],
    PromotedHeaders = Table.PromoteHeaders(RawData, [PromoteAllScalars=true]),

    // Step 2: Sort by HAWB ascending
    SortedRows = Table.Sort(PromotedHeaders, {{"HAWB / Bill of Lading #", Order.Ascending}}),

    // ✅ Step 3a: Add Carrier by VLOOKUP-style join from PBI
    PBI_Carrier = Table.SelectColumns(PBI_Fix, {"#HAWB", "LSP/ Carrier"}),
    RenamedPBI = Table.RenameColumns(PBI_Carrier, {{"#HAWB", "HAWB_PBI"}, {"LSP/ Carrier", "Carrier_Lookup"}}),
    CleanedPBILookup = Table.TransformColumns(RenamedPBI, {{"HAWB_PBI", each Text.Trim(Text.From(_)), type text}}),
    CleanedHAWBKey = Table.TransformColumns(SortedRows, {{"HAWB / Bill of Lading #", each Text.Trim(Text.From(_)), type text}}),
    MergedCarrier = Table.NestedJoin(CleanedHAWBKey, {"HAWB / Bill of Lading #"}, CleanedPBILookup, {"HAWB_PBI"}, "Carrier_Match", JoinKind.LeftOuter),
    AddCarrier = Table.ExpandTableColumn(MergedCarrier, "Carrier_Match", {"Carrier_Lookup"}, {"Carrier"}),

    // Step 3b: Add Total HAWB# fixed to 1
    AddTotalHAWB = Table.AddColumn(AddCarrier, "Total HAWB#", each 1, Int64.Type),

    // ✅ Step 3c: Split null & valid rows for Unique HAWB# calculation
    SplitValidRows = Table.SelectRows(AddTotalHAWB, each [File Item No] <> null),
    SplitNullRows = Table.SelectRows(AddTotalHAWB, each [File Item No] = null),

    AddIndex = Table.AddIndexColumn(SplitValidRows, "Index", 0, 1, Int64.Type),
    Grouped = Table.Group(AddIndex, {"File Name"}, {
        {"AllData", each _, type table [File Item No=nullable number, Index=number]}
    }),
    AddMaxItemIndex = Table.AddColumn(Grouped, "MaxItemIndex", each 
        let
            allRows = [AllData],
            maxItemNo = List.Max(allRows[File Item No]),
            matchingRows = Table.SelectRows(allRows, each [File Item No] = maxItemNo),
            maxRow = Table.First(matchingRows)
        in
            maxRow[Index]
    ),
    Expanded = Table.ExpandTableColumn(AddMaxItemIndex, "AllData", 
        List.RemoveItems(Table.ColumnNames(AddIndex), {"File Name"})),

    AddUniqueHAWB = Table.AddColumn(Expanded, "Unique HAWB#", each if [Index] = [MaxItemIndex] then 1 else 0, Int64.Type),
    RemoveHelperCols = Table.RemoveColumns(AddUniqueHAWB, {"Index", "MaxItemIndex"}),

    AddNullHAWBCol = Table.AddColumn(SplitNullRows, "Unique HAWB#", each null, Int64.Type),
    Recombined = Table.Combine({RemoveHelperCols, AddNullHAWBCol}),

    // Step 3d: Add empty Remarks column
    AddRemarks = Table.AddColumn(Recombined, "Remarks", each null, type text),

    // Step 4: Prepare PBI table
    PBI_Renamed = Table.RenameColumns(PBI_Fix, {{"#HAWB", "HAWB_PBI"}, {"LSP/ Carrier", "Carrier_PBI"}}),

    // Step 5a: Merge HAWB with PBI
    Merged = Table.NestedJoin(AddRemarks, {"HAWB / Bill of Lading #"}, PBI_Renamed, {"HAWB_PBI"}, "PBI_Match", JoinKind.LeftOuter),
    ExpandedPBI = Table.ExpandTableColumn(Merged, "PBI_Match", {"Carrier_PBI"}, {"In PBI"}),

    // Step 5b: Add PBI_Count
    AddPBICount = Table.AddColumn(ExpandedPBI, "PBI_Count", each if [In PBI] = null then 0 else 1, Int64.Type),

    // Step 6: Clean HAWB fields before merge
    CleanedHAWB = Table.TransformColumns(AddPBICount, {{"HAWB / Bill of Lading #", each Text.Trim(Text.From(_)), type text}}),
    WD_Renamed = Table.RenameColumns(WD_Fix, {{"HAWB Number", "HAWB_WD"}}),
    CleanedWD = Table.TransformColumns(WD_Renamed, {{"HAWB_WD", each Text.Trim(Text.From(_)), type text}}),

    MergeWD = Table.NestedJoin(CleanedHAWB, {"HAWB / Bill of Lading #"}, CleanedWD, {"HAWB_WD"}, "WD_Match", JoinKind.LeftOuter),
    AddWDFlag = Table.AddColumn(MergeWD, "WD_Flag", each if List.Count([WD_Match]) > 0 then "WD" else null, type text),
    ReplaceRemarksWD = Table.TransformColumns(AddWDFlag, {{"Remarks", each if _ = null then null else _, type text}}),
    CombineRemarksWD = Table.AddColumn(ReplaceRemarksWD, "Remarks_Updated", each if [Remarks] = null then [WD_Flag] else [Remarks], type text),
    RemoveWDCols = Table.RemoveColumns(CombineRemarksWD, {"WD_Match", "WD_Flag", "Remarks"}),
    RenameWDRemarks = Table.RenameColumns(RemoveWDCols, {{"Remarks_Updated", "Remarks"}}),

    // Step 7: Internal Trucking
    AddInternalTrucking = Table.AddColumn(RenameWDRemarks, "Remarks_IT", each 
        let
            safeRemark = try Text.Upper(Text.Trim(Text.From([Remarks]))) otherwise "",
            safeVendor = try Text.Upper(Text.Trim(Text.From([Freight Supplier]))) otherwise ""
        in
            if safeRemark = "" and safeVendor = "VENDOR" then "Internal Trucking" else safeRemark,
        type text),

    // Step 8: Supplier Paid
    AddSupplierPaid = Table.AddColumn(AddInternalTrucking, "Remarks_Final", each 
        let
            rawRemark = try [Remarks_IT] otherwise null,
            safeRemark = if rawRemark is text then Text.Upper(Text.Trim(rawRemark)) else "",
            rawInco = try [Inco Term] otherwise null,
            safeInco = if rawInco is text then Text.Upper(Text.Trim(rawInco)) else "",
            rawSupp = try [Supplier Name] otherwise null,
            safeSupp = if rawSupp is text then Text.Upper(Text.Trim(rawSupp)) else ""
        in
if safeRemark = "" 
    and (safeInco <> "EXW" and not Text.StartsWith(safeInco, "F"))
    and (not Text.Contains(safeSupp, "INTEL"))
then "Supplier Paid"
else safeRemark,

        type text),

    // Finalize Remarks column
    CleanRemarks = Table.RemoveColumns(AddSupplierPaid, {"Remarks", "Remarks_IT"}),
    FinalRemarksRenamed = Table.RenameColumns(CleanRemarks, {{"Remarks_Final", "Remarks"}}),

    // Step 3g: Reorder columns (after HAWB)
    OriginalCols = Table.ColumnNames(PromotedHeaders),
    NewCols = {"Carrier", "Total HAWB#", "Unique HAWB#", "Remarks", "In PBI", "PBI_Count"},
    Pos = List.PositionOf(OriginalCols, "HAWB / Bill of Lading #"),
    Before = List.FirstN(OriginalCols, Pos + 1),
    After = List.Skip(OriginalCols, Pos + 1),
    NewOrder = Before & NewCols & After,
    FinalReordered = Table.ReorderColumns(FinalRemarksRenamed, NewOrder),

    // ✅ 마지막 빈 행 제거
    CleanResult = Table.SelectRows(FinalReordered, each List.NonNullCount(Record.FieldValues(_)) > 0),

    // ✅ 전체 결과 출력
    Result = CleanResult,
    #"정렬된 행 수" = Table.Sort(Result,{{"File Name", Order.Ascending}})

in
    #"정렬된 행 수"
