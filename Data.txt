let
    // ✅ 외부 쿼리 안전하게 캐싱 (Firewall 우회)
    PBI_Fix = PBI,
    WD_Fix = WD_PBI,

    // Load source data
    Source = Excel.Workbook(File.Contents("C:\Users\ADMIN\Downloads\Reporte Febrero 2025-Rawdata.xlsx"), null, true),
    RawData = Source{[Item="Data",Kind="Sheet"]}[Data],
    PromotedHeaders = Table.PromoteHeaders(RawData, [PromoteAllScalars=true]),

    // Step 2: Sort by HAWB ascending
    SortedRows = Table.Sort(PromotedHeaders, {{"HAWB / Bill of Lading #", Order.Ascending}}),

    // Step 3a: Add Carrier
    AddCarrier = Table.AddColumn(SortedRows, "Carrier", each [Freight Supplier], type text),

    // Step 3b: Add Total HAWB# fixed to 1
    AddTotalHAWB = Table.AddColumn(AddCarrier, "Total HAWB#", each 1, Int64.Type),

    // Step 3c: Add Unique HAWB# based on max File Item No in each File Name
    AddIndex = Table.AddIndexColumn(AddTotalHAWB, "Index", 0, 1, Int64.Type),
    Grouped = Table.Group(AddIndex, {"File Name"}, {
        {"AllData", each _, type table [File Item No=nullable number, Index=number]}
    }),
    AddMaxItemIndex = Table.AddColumn(Grouped, "MaxItemIndex", each 
        let
            allRows = [AllData],
            maxItemNo = List.Max(allRows[File Item No]),
            matchingRows = Table.SelectRows(allRows, each [File Item No] = maxItemNo),
            maxRow = Table.First(matchingRows)
        in
            maxRow[Index]
    ),
    Expanded = Table.ExpandTableColumn(AddMaxItemIndex, "AllData", 
        List.RemoveItems(Table.ColumnNames(AddIndex), {"File Name"})),

    AddUniqueHAWB = Table.AddColumn(Expanded, "Unique HAWB#", each if [Index] = [MaxItemIndex] then 1 else 0, Int64.Type),
    RemoveHelperCols = Table.RemoveColumns(AddUniqueHAWB, {"Index", "MaxItemIndex"}),

    // Step 3d: Add empty Remarks column
    AddRemarks = Table.AddColumn(RemoveHelperCols, "Remarks", each null, type text),

    // Step 4: Prepare PBI table
    PBI_Renamed = Table.RenameColumns(PBI_Fix, {{"#HAWB", "HAWB_PBI"}, {"LSP/ Carrier", "Carrier_PBI"}}),

    // Step 5a: Merge HAWB with PBI
    Merged = Table.NestedJoin(AddRemarks, {"HAWB / Bill of Lading #"}, PBI_Renamed, {"HAWB_PBI"}, "PBI_Match", JoinKind.LeftOuter),
    ExpandedPBI = Table.ExpandTableColumn(Merged, "PBI_Match", {"Carrier_PBI"}, {"In PBI"}),

    // Step 5b: Add PBI_Count
    AddPBICount = Table.AddColumn(ExpandedPBI, "PBI_Count", each if [In PBI] = null then 0 else 1, Int64.Type),

    // Step 6: Clean HAWB fields before merge (Confused - Skipped)
    CleanedHAWB = Table.TransformColumns(AddPBICount, {{"HAWB / Bill of Lading #", each Text.Trim(Text.From(_)), type text}}),
    WD_Renamed = Table.RenameColumns(WD_Fix, {{"HAWB Number", "HAWB_WD"}}),
    CleanedWD = Table.TransformColumns(WD_Renamed, {{"HAWB_WD", each Text.Trim(Text.From(_)), type text}}),

    MergeWD = Table.NestedJoin(CleanedHAWB, {"HAWB / Bill of Lading #"}, CleanedWD, {"HAWB_WD"}, "WD_Match", JoinKind.LeftOuter),
    UpdateRemarksWD = Table.TransformColumns(
        Table.AddColumn(MergeWD, "WD_Flag", each List.Count([WD_Match]) > 0),
        {"WD_Flag", each if _ then "WD" else null, type text}
    ),
    MergeRemarks = Table.AddColumn(UpdateRemarksWD, "Updated Remarks", each if [Remarks] = null then [WD_Flag] else [Remarks], type text),
    DropHelpers = Table.RemoveColumns(MergeRemarks, {"WD_Flag", "WD_Match", "Remarks"}),
    RenameFinalRemarks = Table.RenameColumns(DropHelpers, {{"Updated Remarks", "Remarks"}}),

    // Step 7: Apply "Internal Trucking" if Remarks is still null and Freight Supplier is VENDOR
    AddFinalRemarks = Table.AddColumn(RenameFinalRemarks, "Final Remarks", each 
        let
            currentRemark = try [Remarks] otherwise null,
            freight = try Text.Upper([Freight Supplier]) otherwise null
        in
            if currentRemark = null and freight = "VENDOR" then "Internal Trucking" else currentRemark,
        type text),
    DropOldRemarks = Table.RemoveColumns(AddFinalRemarks, {"Remarks"}),
    RenameFinal = Table.RenameColumns(DropOldRemarks, {{"Final Remarks", "Remarks"}}),

    // ✅ Step 8: Apply "Supplier Paid" logic
    AddStep8Remarks = Table.AddColumn(RenameFinal, "Final Remarks Step8", each 
        let
            currentRemark = try [Remarks] otherwise null,
            incoterm = try Text.Upper(Text.Trim([Incoterm])) otherwise "",
            supplier = try Text.Upper(Text.Trim([Supplier Name])) otherwise ""
        in
            if (currentRemark = null or currentRemark = "") 
                and (incoterm <> "EXW" and not Text.StartsWith(incoterm, "F"))
                and (not Text.Contains(supplier, "INTEL"))
            then "Supplier Paid" 
            else currentRemark,
        type text),
    DropStep7Remarks = Table.RemoveColumns(AddStep8Remarks, {"Remarks"}),
    RenameStep8Remarks = Table.RenameColumns(DropStep7Remarks, {{"Final Remarks Step8", "Remarks"}}),

    // Step 3g: Reorder columns (after HAWB)
    OriginalCols = Table.ColumnNames(PromotedHeaders),
    NewCols = {"Carrier", "Total HAWB#", "Unique HAWB#", "Remarks", "In PBI", "PBI_Count"},
    Pos = List.PositionOf(OriginalCols, "HAWB / Bill of Lading #"),
    Before = List.FirstN(OriginalCols, Pos + 1),
    After = List.Skip(OriginalCols, Pos + 1),
    NewOrder = Before & NewCols & After,
    FinalReordered = Table.ReorderColumns(RenameStep8Remarks, NewOrder),
    #"필터링된 행" = Table.SelectRows(FinalReordered, each ([Remarks] = "Supplier Paid"))

in
    #"필터링된 행"
